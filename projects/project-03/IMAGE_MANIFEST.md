import os
from PIL import Image

BASE = "/Users/takao/Documents/00_Product_Develpment/Substack/Portfolio/intentfirst"
P3   = os.path.join(BASE, "assets/img/p3")
THM  = os.path.join(BASE, "assets/img/thumbs")
AV   = os.path.join(BASE, "assets/img/avatars")
os.makedirs(THM, exist_ok=True)
os.makedirs(AV,  exist_ok=True)

THUMB_SOURCES = [
    "naomi-portrait.png", "marcus-portrait.png", "zara-portrait.png", "james-portrait.png",
    "P3-hero-naomi-train-thumb.png",
    "P3-ch01-setup.png", "P3-ch02-grammar.png", "P3-ch03-day.png",
    "P3-ch04-breaks.png", "P3-ch05-overtime.png", "P3-ch06-close.png",
]

for name in THUMB_SOURCES:
    src = os.path.join(P3, name)
    if not os.path.exists(src):
        print("SKIP missing:", name); continue
    img = Image.open(src).convert("RGB")
    w, h = img.size
    stem = os.path.splitext(name)[0]
    for tw, suffix in [(1600, ""), (800, "-sm")]:
        if tw >= w:
            continue
        nh = int(round(h * tw / w))
        out = os.path.join(THM, f"{stem}{suffix}.png")
        img.resize((tw, nh), Image.LANCZOS).save(out, "PNG", optimize=True)
        print("wrote", out)

# Avatar crops: tight square head+shoulders, 400x400 PNG.
AVATARS = [
    ("naomi-portrait.png",  "naomi-avatar.png"),
    ("marcus-portrait.png", "marcus-avatar.png"),
    ("zara-portrait.png",   "zara-avatar.png"),
    ("james-portrait.png",  "james-avatar.png"),
]
for src_name, out_name in AVATARS:
    src = os.path.join(P3, src_name)
    if not os.path.exists(src):
        print("SKIP missing avatar source:", src_name); continue
    img = Image.open(src).convert("RGB")
    w, h = img.size
    # Portraits are 1:1 (4K square) per the JSON spec, so center crop is the face.
    side = min(w, h)
    left = (w - side) // 2
    top  = max(0, (h - side) // 2 - int(side * 0.05))   # nudge up 5% to favor face over shoulders
    img.crop((left, top, left + side, top + side)).resize((400, 400), Image.LANCZOS) \
       .save(os.path.join(AV, out_name), "PNG", optimize=True)
    print("wrote avatar", out_name)

print("done")
